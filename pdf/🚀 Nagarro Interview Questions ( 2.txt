Hands-On Task in React Js:
16. Build a dynamic multi-select dropdown using React, Handle selection/deselection, Show selected options as tags


import React, { useState, useRef, useEffect } from "react";

const MultiSelectDropdown = ({ options }) => {
  const [selected, setSelected] = useState([]);
  const [isOpen, setIsOpen] = useState(false);
  const [search, setSearch] = useState("");
  const dropdownRef = useRef(null);

  // Filter options
  const filteredOptions = options.filter(
    (opt) => opt.toLowerCase().includes(search.toLowerCase()) && !selected.includes(opt)
  );

  // Handle selection/deselection
  const handleSelect = (option) => {
    setSelected([...selected, option]);
    setSearch("");
  };

  // Remove selected tag
  const removeTag = (option) => {
    setSelected(selected.filter((item) => item !== option));
  };

  // Close dropdown when clicked outside
  useEffect(() => {
    const handleClickOutside = (e) => {
      if (dropdownRef.current && !dropdownRef.current.contains(e.target)) {
        setIsOpen(false);
      }
    };
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  return (
    <div className="relative w-64" ref={dropdownRef}>
      {/* Input box with tags */}
      <div
        className="flex flex-wrap items-center gap-2 border p-2 rounded cursor-text"
        onClick={() => setIsOpen(true)}
      >
        {selected.map((item) => (
          <span
            key={item}
            className="flex items-center gap-1 bg-blue-500 text-white px-2 py-1 rounded-full text-sm"
          >
            {item}
            <button
              onClick={(e) => {
                e.stopPropagation();
                removeTag(item);
              }}
              className="ml-1 text-xs font-bold"
            >
              ×
            </button>
          </span>
        ))}

    
      </div>
            {/* Input field for search */}
        <input
          type="text"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="flex-1 min-w-[80px] outline-none text-sm"
          placeholder={selected.length === 0 ? "Search & select..." : ""}
          onFocus={() => setIsOpen(true)}
        />

      {/* Dropdown list */}
      {isOpen && filteredOptions.length > 0 && (
        <ul className="absolute left-0 right-0 mt-1 max-h-48 overflow-y-auto border rounded bg-white shadow z-10">
          {filteredOptions.map((option) => (
            <li
              key={option}
              onClick={() => handleSelect(option)}
              className="px-3 py-2 cursor-pointer hover:bg-blue-100"
            >
              {option}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
};

// Example usage
export default function App() {
  const fruits = ["Apple", "Banana", "Orange", "Mango", "Pineapple", "Papaya", "Grapes"];
  return (
    <div className="p-8">
      <h1 className="text-xl font-bold mb-4">Searchable Multi-Select Dropdown</h1>
      <MultiSelectDropdown options={fruits} />
    </div>
  );
}
